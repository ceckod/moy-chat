Инструкции:

  1. Отиди на dashboard.cloudflare.com → Workers & Pages → Create Worker

export default {
  async fetch(request) {
    const url = new URL(request.url);
    const target = url.searchParams.get('url');
    if (!target) return new Response('Missing ?url=', { status: 400 });

    // Clone headers, strip problematic ones, add realistic browser headers
    const headers = new Headers(request.headers);
    headers.set('User-Agent', 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 Chrome/126.0.0.0 Safari/537.36');
    headers.set('Accept', 'text/html,application/xhtml+xml,application/xml;q=0.9,image/webp,*/*;q=0.8');
    headers.set('Accept-Language', 'en-US,en;q=0.9');
    headers.set('Referer', 'https://www.google.com/');
    headers.delete('Origin');
    headers.delete('CF-Access-Client-Id');
    headers.delete('CF-Access-Client-Secret');

    const response = await fetch(target, {
      method: request.method,
      headers: headers,
      redirect: 'follow',
    });

    // CORS headers
    const corsHeaders = {
      'Access-Control-Allow-Origin': '*',
      'Access-Control-Allow-Methods': 'GET, POST, OPTIONS',
      'Access-Control-Allow-Headers': '*',
      'Access-Control-Max-Age': '86400',
    };

    // For HTML pages, inject a script to extract video/audio sources
    const contentType = response.headers.get('content-type') || '';
    let body = response.body;
    
    if (contentType.includes('text/html')) {
      const text = await response.text();
      const injected = `<script>
        (function() {
          const sources = [];
          document.querySelectorAll('video source, video, audio source, audio, [data-url], iframe').forEach(el => {
            const src = el.src || el.dataset.url || el.href;
            if (src) sources.push(src);
          });
          const meta = {
            title: document.title,
            url: window.location.href,
            sources: sources,
            videoCount: document.querySelectorAll('video').length,
            canvasCount: document.querySelectorAll('canvas').length,
          };
          // Store for postMessage
          window.__vizMeta = meta;
          // Also try to extract from scripts
          document.querySelectorAll('script').forEach(s => {
            const c = s.textContent || '';
            const matches = c.match(/https?:\\/\\/[^\"'\\s]+(?:\\.mp4|\\.webm|\\/video\\/|\\/stream\\/)[^\"'\\s]*/g);
            if (matches) meta.sources.push(...matches);
          });
          console.log('VizMeta:', JSON.stringify(meta));
        })();
      </script></body>`;
      body = text.replace('</body>', injected);
      body = new Blob([body], { type: 'text/html' });
    }

    const responseHeaders = new Headers(response.headers);
    Object.entries(corsHeaders).forEach(([k,v]) => responseHeaders.set(k,v));
    responseHeaders.set('Content-Type', contentType);

    return new Response(body, {
      status: response.status,
      headers: responseHeaders,
    });
  }
}
